# ==============================================================================
# Примеры форматов прокси для KROTTUN (пакетный импорт или одиночное добавление)
# ==============================================================================

# Формат 1: IP:PORT:USER:PASS
192.0.2.1:1080:demo_user:secret_pass

# Формат 2: IP:PORT (без авторизации)
192.0.2.2:8080

# Формат 3: URL-схема SOCKS5
socks5://demo_user:secret_pass@192.0.2.3:1080

# Формат 4: URL-схема HTTP / HTTPS
http://demo_user:secret_pass@192.0.2.4:8080
